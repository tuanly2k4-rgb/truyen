    <script>

        // =========================================================

        // 1. CẤU HÌNH GITHUB CỦA BẠN TẠI ĐÂY

        // =========================================================

        const GITHUB_USER = 'tuanly2k4-rgb'; // VD: 'nguyenvana'

        const GITHUB_REPO = 'truyen';   // VD: 'kho-truyen'

        // =========================================================


        const appContainer = document.getElementById('app-container');

        const loadingIndicator = document.getElementById('loading');


        function showLoading(show) {

            loadingIndicator.style.display = show ? 'block' : 'none';

        }


        function showError(message) {

            appContainer.innerHTML = `

                <div class="error">

                    <strong>Không thể tải dữ liệu:</strong> ${message}<br><br>

                    <i>Hướng dẫn: Hãy chắc chắn bạn đã thay đổi biến GITHUB_USER và GITHUB_REPO trong mã nguồn, và kho chứa (Repository) của bạn đang ở chế độ Public.</i>

                </div>`;

        }


        async function fetchStories() {

            showLoading(true);

            try {

                const response = await fetch(`https://api.github.com/repos/${GITHUB_USER}/${GITHUB_REPO}/contents/`);

                

                if (!response.ok) {

                    if (response.status === 404) throw new Error("Không tìm thấy Kho chứa (Repository). Bạn đã điền đúng tên chưa?");

                    if (response.status === 403) throw new Error("Bị giới hạn lượt truy cập API từ Github, vui lòng thử lại sau.");

                    throw new Error(`Mã lỗi: ${response.status}`);

                }


                const data = await response.json();

                

                // Lọc lấy các thư mục (loại bỏ các file ngoài cùng như README.md nếu có)

                const folders = data.filter(item => item.type === 'dir');

                renderHome(folders);


            } catch (error) {

                showError(error.message);

            } finally {

                showLoading(false);

            }

        }


        async function fetchChapters(folderName) {

            showLoading(true);

            try {

                const response = await fetch(`https://api.github.com/repos/${GITHUB_USER}/${GITHUB_REPO}/contents/${encodeURIComponent(folderName)}`);

                

                if (!response.ok) throw new Error('Không thể lấy nội dung thư mục');


                const data = await response.json();

                const files = data.filter(item => item.type === 'file');

                renderStoryInfo(folderName, files);


            } catch (error) {

                showError(error.message);

            } finally {

                showLoading(false);

            }

        }


        async function fetchFileContent(fileUrl, fileName, folderName) {

            showLoading(true);

            try {

                const response = await fetch(fileUrl);

                if (!response.ok) throw new Error('Không thể tải file nội dung');

                

                const text = await response.text();

                renderReading(fileName, text, folderName);

            } catch (error) {

                showError(error.message);

            } finally {

                showLoading(false);

            }

        }


        function renderHome(folders) {

            if (folders.length === 0) {

                appContainer.innerHTML = '<p style="text-align:center; margin-top:50px;">Chưa có tác phẩm nào được đăng tải.</p>';

                return;

            }


            let html = '<h2>Danh sách Tác Phẩm</h2><ul>';

            folders.forEach(folder => {

                html += `<li><a onclick="fetchChapters('${folder.name.replace(/'/g, "\\'")}')">${folder.name}</a></li>`;

            });

            html += '</ul>';


            appContainer.innerHTML = html;

            window.scrollTo(0, 0);

        }


        function renderStoryInfo(folderName, files) {

            let html = `<button class="back-btn" onclick="goHome()">← Trở về trang chủ</button>`;

            html += `<h2>${folderName}</h2>`;


            if (files.length === 0) {

                html += '<p>Tác phẩm này hiện chưa có chương nào.</p>';

            } else {

                html += '<ul>';

                files.forEach(file => {

                    html += `<li><a onclick="fetchFileContent('${file.download_url}', '${file.name.replace(/'/g, "\\'")}', '${folderName.replace(/'/g, "\\'")}')">${file.name.replace(/\.[^/.]+$/, "")}</a></li>`;

                });

                html += '</ul>';

            }


            appContainer.innerHTML = html;

            window.scrollTo(0, 0);

        }


        function renderReading(fileName, content, folderName) {

            let html = `<button class="back-btn" onclick="fetchChapters('${folderName.replace(/'/g, "\\'")}')">← Trở về danh sách chương</button>`;

            html += `<h2>${fileName.replace(/\.[^/.]+$/, "")}</h2>`;

            html += `<div class="content-box">${content}</div>`;

            appContainer.innerHTML = html;

            window.scrollTo(0, 0);

        }


        window.goHome = function() {

            fetchStories();

        }


        // Bắt đầu khởi chạy trang web

        goHome();
